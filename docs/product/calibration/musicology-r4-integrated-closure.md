function runK16Finite(){
 const eq=(a,b)=>JSON.stringify(a)===JSON.stringify(b);
 const cat={kind:"catalog",id:"SFW40492",recorded:2000,released:2003,composed:null,place:"Rangoon",tracks:9,source:"S576",audioRead:false};
